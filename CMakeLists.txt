cmake_minimum_required(VERSION 3.25)
project(fuc VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
find_package(CURL 7.85 REQUIRED)
find_package(yaml-cpp 0.8 CONFIG REQUIRED)
find_package(nlohmann_json 3.10 CONFIG REQUIRED)
add_library(fuc_core src/client.cpp src/commands.cpp)
target_include_directories(fuc_core PUBLIC include)
target_link_libraries(fuc_core PUBLIC CURL::libcurl yaml-cpp::yaml-cpp nlohmann_json::nlohmann_json)
target_compile_options(fuc_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(fuc src/main.cpp)
target_link_libraries(fuc PRIVATE fuc_core)
target_compile_options(fuc PRIVATE -Wall -Wextra -Wpedantic)
include(CTest)
if(BUILD_TESTING)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_test(NAME integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.py $<TARGET_FILE:fuc>)
endif()
install(TARGETS fuc RUNTIME DESTINATION bin)
